📜 SQL Schema Proyek Supabase

Buka dashboard Supabase Anda, masuk ke **SQL Editor**, buat query baru, lalu jalankan query SQL berikut untuk membuat struktur database relasional Angkringan Mas Eja:

-- Query SQL Relasional Angkringan Mas Eja create table menu_items ( id bigint primary key generated always as identity, name text not null, category text not null, price numeric not null, harga_modal numeric not null default 0, stok_sistem integer not null default 0, stok_minimum integer not null default 5, emoji text, description text, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); create table transactions ( id bigint primary key generated always as identity, tanggal date not null default current_date, jam time not null default current_time, items jsonb not null, total_harga numeric not null, pembayaran text not null, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); create table bon_hutang ( id bigint primary key generated always as identity, nama_pelanggan text not null, no_whatsapp text, tanggal_bon date not null default current_date, jumlah_hutang numeric not null, sisa_hutang numeric not null, status_lunas boolean not null default false, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); create table expenses ( id bigint primary key generated always as identity, tanggal date not null default current_date, keterangan text not null, jumlah numeric not null, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); create table food_waste ( id bigint primary key generated always as identity, tanggal date not null default current_date, menu_item_id bigint references menu_items(id), qty_basi integer not null, kerugian_modal numeric not null, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- Kebijakan RLS Akses Publik (Buka untuk Anon Key) alter table menu_items enable row level security; alter table transactions enable row level security; alter table bon_hutang enable row level security; alter table expenses enable row level security; alter table food_waste enable row level security; create policy "menu_select" on menu_items for select using (true); create policy "menu_all" on menu_items for all using (true); create policy "trans_all" on transactions for all using (true); create policy "bon_all" on bon_hutang for all using (true); create policy "exp_all" on expenses for all using (true); create policy "waste_all" on food_waste for all using (true);
